import random

class Traveler:
    def __init__(self, name):
        self.name = name
        self.hunting = random.randint(3, 7)   # навык охоты (влияет на добычу)
        self.endurance = random.randint(3, 7) # выносливость (сопротивление цинге)
        self.alive = True
        self.scurvy = False  # болеет ли цингой

class Settlement:
    def __init__(self):
        self.travelers = [Traveler("Ермак"), Traveler("Иван")]
        self.fur = 50          # пушнина (метан)
        self.equipment = 0     # экипировка (говно)
        self.money = 100       # деньги (быдло)
        self.dogs = 0          # собаки и лошади (роботы)
        self.horses = 0
        self.lands = 1         # открытые земли (матан) – базовый регион
        self.charters = 0      # царские грамоты
        self.cabbage = 0       # квашеная капуста
        self.church = False    # храм построен?
        self.palisade = False  # частокол построен?
        self.gnaw_charters = 0 # гневные грамоты
        self.year = 1
        self.season = 0        # 0-весна, 1-лето, 2-осень, 3-зима
        self.fur_produced_this_cycle = 0  # для проверки ясака
        self.years_since_check = 0
        self.total_fur_sent_to_tsar = 0   # для достижения 100 грамот
        self.penalty_next_season = 0      # штраф за пьянство (процент снижения добычи)
        self.turn = 0

    def total_animals(self):
        return self.dogs + self.horses

    def max_travelers(self):
        return len(self.travelers)

    def living_travelers(self):
        return [t for t in self.travelers if t.alive]

    def count_scurvy(self):
        return sum(1 for t in self.travelers if t.alive and t.scurvy)

    def has_scurvy(self):
        return self.count_scurvy() > 0

    def remove_dead(self):
        self.travelers = [t for t in self.travelers if t.alive]

    def add_traveler(self):
        # редко, но можно нанять за деньги (или как бонус)
        if self.money >= 50:
            self.money -= 50
            new_t = Traveler("Новобранец")
            self.travelers.append(new_t)
            return True
        return False

    def run(self):
        print("="*60)
        print("Добро пожаловать в игру «Первопроходец Сибири»!")
        print("Вы – исследователь и землепроходец, осваивающий сибирские земли.")

            self.display_status()
            if not self.check_game_over():
                break
            cmd = input("> ").strip().lower()
            if not cmd:
                continue
            parts = cmd.split()
            action = parts[0]
            args = parts[1:]
            if action in self.commands:
                try:
                    self.commands[action](args)
                except Exception as e:
                    print("Ошибка выполнения команды:", e)
            else:
                print("Неизвестная команда. Введите 'помощь' для списка.")
            # После каждого действия проверяем состояние
            if not self.running:
                break
            self.after_action()

    def advance_season(self):
        s = self.settlement
        # Переход к следующему сезону
        s.season = (s.season + 1) % 4
        if s.season == 0:  # новая весна – новый год
            s.year += 1
            s.years_since_check += 1
            # весной приезжают купцы, цена случайная
            self.merchant_price = random.randint(2, 10)
            # Проверка ясака каждые 3 года
            if s.years_since_check >= 3:
                self.check_yasak()

        # Зимой – расход ресурсов
        if s.season == 3:  # зима
            self.winter_consumption()

        # Проверка пьянства (в конце каждого сезона)
        self.check_drunkard()

        # Случайные события
        self.event_manager.random_event()

    def winter_consumption(self):
        s = self.settlement
        # Зимой каждый живущий путешественник потребляет экипировку и пушнину
        need_equip = len(s.living_travelers()) * 2
        need_fur = len(s.living_travelers()) * 1
        # Также животные едят
        need_equip += s.total_animals() * 2
        need_fur += s.total_animals() * 1

        if s.equipment < need_equip or s.fur < need_fur:
            # Не хватает – голод, смерть
            deficit_equip = max(0, need_equip - s.equipment)
            deficit_fur = max(0, need_fur - s.fur)
            # Умирают случайные путешественники (по одному за каждые 5 недостающих единиц)
            deaths = (deficit_equip + deficit_fur) // 5
            if deaths > 0:
                living = s.living_travelers()
                for _ in range(min(deaths, len(living))):
                    victim = random.choice(living)
                    victim.alive = False
                    living.remove(victim)
                print(f"❄️ Зимний голод! Умерло {deaths} путешественников.")
            # Тратим то, что есть
            s.equipment = max(0, s.equipment - need_equip)
            s.fur = max(0, s.fur - need_fur)
        else:
            s.equipment -= need_equip
            s.fur -= need_fur
            print("❄️ Зима прошла благополучно, ресурсов хватило.")

        # Лечение цинги капустой
        if s.cabbage > 0 and s.has_scurvy():
            sick = [t for t in s.travelers if t.alive and t.scurvy]
            for t in sick[:s.cabbage]:
                t.scurvy = False
                s.cabbage -= 1
                print(f"🥬 {t.name} вылечен от цинги квашеной капустой.")
            if s.has_scurvy():
                print("⚠️ Не хватило капусты, некоторые путешественники всё ещё больны.")

        # Убираем мёртвых
        s.remove_dead()

    def check_drunkard(self):
        s = self.settlement
        # Порог пьянства: 2000 (или 3000 если есть храм)
        threshold = 3000 if s.church else 2000
        if s.money > threshold:
            if s.charters > 0:
                s.charters -= 1
                s.money -= 500  # штраф на пиры
                print(f"🍺 Пьянство и разгул! Царь отнял 1 грамоту. Осталось грамот: {s.charters}")
            else:
                # Нет грамот – штраф к добыче на следующий сезон
                s.penalty_next_season += 30  # -30% к добыче
                print("🍺 Пьянство и лень! Путешественники будут добывать меньше пушнины в следующем сезоне.")

                # Можно не трогать, но штраф будет действовать

    def check_yasak(self):
        s = self.settlement
        # Каждые 3 года нужно сдать 200 пушнины (ясачная повинность)
        required = 200
        if s.fur >= required:
            s.fur -= required
            s.charters += 1
            s.gnaw_charters = 0  # сброс гневных
            print(f"👑 Ясак выполнен! Царь жалует грамоту. Всего грамот: {s.charters}")
        else:
            s.gnaw_charters += 1
            print(f"⚠️ Ясак не выполнен! Царь гневается. Гневных грамот: {s.gnaw_charters}")
            if s.gnaw_charters >= 3:
                print("💀 Три гневные грамоты! Царь ссылает вас в Охотскую тюрьму. Поражение!")
                self.running = False
        s.years_since_check = 0

    def check_game_over(self):
        s = self.settlement
        if len(s.living_travelers()) == 0:
            print("💀 Все путешественники погибли. Игра окончена.")
            self.running = False
            return False
        if s.fur <= 0 and s.equipment <= 0 and s.season == 3:
            # Зимой нечем кормить – голод
            print("💀 Ресурсы на исходе зимой – все погибли от голода.")
            self.running = False
            return False
        if s.gnaw_charters >= 3:
            print("💀 Три гневные грамоты – поражение.")
            self.running = False
            return False
        # Проверка победы
        if s.charters >= 5:
            print("🎉 Поздравляем! Вы накопили 5 царских грамот и успешно освоили Сибирь! Победа!")
            self.running = False
            return False
        if s.charters >= 100:
            print("🏙️ ВЕЛИКАЯ ПОБЕДА! В вашу честь назван город! Вы вошли в историю как величайший первопроходец!")
            self.running = False
            return False
        return True

        print(f"Путешественников: {len(s.living_travelers())} (всего {len(s.travelers)})")
        if s.has_scurvy():

        print(f"Пушнина: {s.fur} | Экипировка: {s.equipment} | Деньги: {s.money} руб.")
        print(f"Собаки/лошади: {s.dogs}/{s.horses} (всего {s.total_animals()})")
        print(f"Открытые земли: {s.lands} | Царские грамоты: {s.charters} | Гневные: {s.gnaw_charters}")
        if s.church:
            print("🏛️ Храм построен (порог пьянства +1000)")
        if s.palisade:
            print("🛡️ Частокол построен (защита от разбойников)")
        if s.cabbage > 0:
            print(f"🥬 Квашеная капуста: {s.cabbage}")
        if s.penalty_next_season > 0:
            print(f"⚠️ Штраф к добыче: -{s.penalty_next_season}%")
        if s.season == 0:
            print(f"💰 Цена пушнины у купцов: {self.merchant_price} руб./ед.")
        print("-"*50)

    def cmd_skip(self, args):
        # переход к следующему сезону
        self.advance_season()
        # после пропуска проверяем поражение/победу
        if self.running:
            self.check_game_over()

    def cmd_sell(self, args):
        if self.settlement.season != 0:
            print("❌ Купцы приезжают только весной!")
            return
        if not args:
            print("Укажите количество пушнины для продажи.")
            return
        try:
            amount = int(args[0])
        except:
            print("Неверное число.")
            return
        s = self.settlement
        if amount <= 0:
            print("Количество должно быть положительным.")
            return
        if amount > s.fur:
            print(f"У вас только {s.fur} пушнины.")
            return
        s.fur -= amount
        revenue = amount * self.merchant_price
        # Добавляем бонус за грамоты: каждая грамота даёт +10% к цене
        bonus = 1 + 0.1 * s.charters
        revenue = int(revenue * bonus)
        s.money += revenue
        print(f"✅ Продано {amount} пушнины за {revenue} рублей.")

        print(f"👑 Передано {amount} пушнины царю. Получено {earned} грамот. Всего грамот: {s.charters}.")

    def cmd_expedition(self, args):
        if self.settlement.season != 1:
            print("❌ Экспедиции возможны только летом!")
            return
        if len(args) < 2:
            print("Использование: отправить <кол-во путешественников> <регион>")
            print("Регион: 'новый' или число от 1 до", self.settlement.lands)
            return
        try:
            count = int(args[0])
        except:
            print("Неверное число.")
            return
        region = args[1]
        s = self.settlement
        living = s.living_travelers()
        if count > len(living):
            print(f"У вас только {len(living)} живых путешественников.")
            return
        if count <= 0:
            print("Нужно отправить хотя бы одного.")
            return
        # Проверка экипировки
        need_equip = count * 3  # базовый расход
        if s.equipment < need_equip:
            print(f"Недостаточно экипировки. Нужно {need_equip}, есть {s.equipment}.")
            return
        # Проверка, новый ли регион
        is_new = (region == 'новый')
        if is_new:
            # открываем новую землю (увеличиваем счётчик)
            region_idx = s.lands + 1
        else:
            try:
                region_idx = int(region)
            except:
                print("Регион должен быть 'новый' или число.")
                return
            if region_idx < 1 or region_idx > s.lands:
                print(f"Доступны регионы от 1 до {s.lands} или 'новый'.")
                return
        # Расход экипировки
        s.equipment -= need_equip
        # Определяем базовую добычу: зависит от навыков, количества, региона
        total_hunting = sum(t.hunting for t in living[:count])
        # Бонус за открытые земли (каждая земля даёт +10% к добыче)
        land_bonus = 1 + 0.1 * s.lands
        # Штраф за пьянство
        penalty = 1 - s.penalty_next_season / 100.0
        if penalty < 0:
            penalty = 0
        # Бонус за собак и лошадей: каждая дает +5%
        animal_bonus = 1 + 0.05 * s.total_animals()
        # Случайный фактор (от 0.8 до 1.2)
        random_factor = random.uniform(0.8, 1.2)
        # Добыча
        base_fur = total_hunting * 2  # базово
        fur_gained = int(base_fur * land_bonus * animal_bonus * penalty * random_factor)
        if fur_gained < 0:
            fur_gained = 0
        # Если открываем новую землю, добавляем бонус к матану (lands)
        if is_new:
            s.lands += 1
            print(f"🗺️ Открыта новая земля! Теперь открытых земель: {s.lands}")
        # Проверка на цингу
        sick_chance = 0.1  # 10% на каждого
        sick_count = 0
        for t in living[:count]:
            if random.random() < sick_chance:
                # Выносливость снижает шанс заболеть
                if random.random() > t.endurance / 10.0:
                    t.scurvy = True
                    sick_count += 1
        if sick_count > 0:
            print(f"🤒 {sick_count} путешественников заболели цингой в походе.")
        # Проверка на разбойников
        bandit_chance = 0.2 if not s.palisade else 0.1
        if random.random() < bandit_chance:
            # Нападение
            stolen = fur_gained // 3  # крадут треть
            if stolen > 0:
                fur_gained -= stolen
                print(f"🏴 Разбойники украли {stolen} пушнины!")
            # Могут убить путешественников
            if random.random() < 0.3:
                victims = random.sample(living[:count], min(1, count))
                for v in victims:
                    v.alive = False
                print(f"⚔️ Разбойники убили {len(victims)} путешественника(ов).")
                s.remove_dead()
        else:
            print("✅ Поход прошёл без нападений.")
        # Добавляем пушнину
        s.fur += fur_gained
        s.fur_produced_this_cycle += fur_gained
        print(f"🦊 Добыто {fur_gained} пушнины.")
        # Сброс штрафа за пьянство (действовал только один сезон)
        s.penalty_next_season = 0

    def cmd_buy_equipment(self, args):

        s.money -= cost

        s.palisade = True
        print("🛡️ Частокол построен! Риск нападений разбойников снижен.")

    def cmd_bribe_bandits(self, args):
        if not args:
            print("Укажите сумму для подкупа.")
            return
        try:
            amount = int(args[0])
        except:
            print("Неверное число.")
            return
        s = self.settlement
        if amount > s.money:
            print("Недостаточно денег.")
            return
        s.money -= amount
        # Подкуп снижает активность разбойников на этот год (временно)
        # Установим флаг, который уменьшит шанс нападения в этом сезоне
        # Просто сделаем так: вероятность нападения в этом лете снижается
        self.bandit_activity = max(0.05, self.bandit_activity - 0.1)
        print(f"🤝 Разбойники подкуплены на {amount} руб. Активность снижена.")

class EventManager:
    def __init__(self, game):
        self.game = game

    def random_event(self):
        # Случайные события, которые могут произойти в любой сезон (кроме, может, зимы)
        # Вероятность 30%
        if random.random() > 0.3:
            return
        s = self.game.settlement
        events = [
            self.event_find_fur,
            self.event_find_equipment,
            self.event_merchant_cheat,
            self.event_blizzard,
            self.event_animal_rampage,
        ]
        choice = random.choice(events)
        choice()

    def event_find_fur(self):
        # Находка богатого соболиного гнезда
        bonus = random.randint(10, 40)
        self.game.settlement.fur += bonus
        print(f"🎉 Случайная находка: обнаружено богатое соболиное гнездо! +{bonus} пушнины.")

    def event_find_equipment(self):
        # Находка экипировки
        bonus = random.randint(5, 20)
        self.game.settlement.equipment += bonus
        print(f"🎉 Найдена старая кладовая! +{bonus} экипировки.")

    def event_merchant_cheat(self):
        # Купец обманывает – теряем деньги
        if self.game.settlement.money > 20:
            loss = random.randint(10, 50)
            self.game.settlement.money -= loss
            print(f"😤 Купец обвесил вас! Потеряно {loss} рублей.")
        else:
            print("Купец попытался обмануть, но у вас слишком мало денег – он ушёл ни с чем.")

    def event_blizzard(self):
        # Снежная буря зимой – дополнительный расход экипировки
        if self.game.settlement.season == 3:
            loss = random.randint(5, 15)
            self.game.settlement.equipment = max(0, self.game.settlement.equipment - loss)
            print(f"❄️ Снежная буря! Потеряно {loss} экипировки.")

    def event_animal_rampage(self):
        # Животные взбесились – теряем часть собак/лошадей
        s = self.game.settlement
        if s.total_animals() > 0:
            loss = random.randint(1, min(3, s.total_animals()))
            # Убираем сначала собак, потом лошадей
            for _ in range(loss):
                if s.dogs > 0:
                    s.dogs -= 1
                elif s.horses > 0:
                    s.horses -= 1
            print(f"🐾 Животные взбесились! Потеряно {loss} голов скота.")
